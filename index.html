""" Airliner with animated RETRACTABLE LANDING GEAR -- Blender Python script HOW TO RUN 1. Open Blender (3.x or 4.x), switch to the "Scripting" workspace. 2. Text > Open > pick this file, then press "Run Script" (Alt+P). 3. Go back to the "Layout" workspace, press SPACE to play the animation. ANIMATION (24 fps timeline, frames 1-160) 1-20 gear DOWN and locked 20-65 gear RETRACTS (nose gear folds forward, main gears fold inward) 65-105 gear UP 105-145 gear EXTENDS again 145-160 gear DOWN and locked Coordinates: +X = nose, +Y = left wing, +Z = up. 1 unit ~ 1 metre-ish. """ import math import bpy import bmesh from mathutils import Vector # ----------------------------------------------------------------- settings CLEAR_SCENE = True # delete everything in the scene before building ADD_GROUND = True # grey ground plane touching the wheels GROUND_Z = -2.6 # height of the ground (= bottom of the tires) SAVE_BLEND_TO = "" # e.g. "C:/Users/you/Desktop/aircraft.blend" (empty = don't save) FRAME_START, FRAME_END = 1, 160 # ------------------------------------------------------------ scene cleanup if bpy.context.object and bpy.context.object.mode != 'OBJECT': bpy.ops.object.mode_set(mode='OBJECT') if CLEAR_SCENE: for o in list(bpy.data.objects): bpy.data.objects.remove(o, do_unlink=True) old = bpy.data.collections.get("Aircraft") if old: for o in list(old.objects): bpy.data.objects.remove(o, do_unlink=True) bpy.data.collections.remove(old) coll = bpy.data.collections.new("Aircraft") bpy.context.scene.collection.children.link(coll) # ---------------------------------------------------------------- materials def make_mat(name, color, metallic=0.0, roughness=0.5): mat = bpy.data.materials.new(name) mat.use_nodes = True bsdf = mat.node_tree.nodes.get("Principled BSDF") if bsdf: bsdf.inputs["Base Color"].default_value = color bsdf.inputs["Metallic"].default_value = metallic bsdf.inputs["Roughness"].default_value = roughness mat.diffuse_color = color # viewport color mat.metallic = metallic mat.roughness = roughness return mat M_BODY = make_mat("Body_White", (0.92, 0.93, 0.95, 1), 0.0, 0.35) M_WING = make_mat("Wing_Grey", (0.62, 0.65, 0.70, 1), 0.3, 0.4) M_TAIL = make_mat("Tail_Blue", (0.04, 0.18, 0.55, 1), 0.0, 0.35) M_GLASS = make_mat("Glass_Dark", (0.02, 0.04, 0.08, 1), 0.0, 0.1) M_ENGINE = make_mat("Engine_Grey", (0.45, 0.47, 0.50, 1), 0.6, 0.35) M_METAL = make_mat("Gear_Metal", (0.75, 0.77, 0.80, 1), 0.9, 0.3) M_TIRE = make_mat("Tire_Black", (0.02, 0.02, 0.02, 1), 0.0, 0.8) M_GROUND = make_mat("Ground", (0.22, 0.25, 0.28, 1), 0.0, 0.9) # ------------------------------------------------------------------ helpers def register(obj, parent=None, mat=None, smooth=False): """Move obj into the Aircraft collection, parent it, assign material.""" for c in list(obj.users_collection): c.objects.unlink(obj) coll.objects.link(obj) if parent is not None: obj.parent = parent if mat is not None and obj.type == 'MESH': obj.data.materials.append(mat) if smooth and obj.type == 'MESH': for p in obj.data.polygons: p.use_smooth = True return obj def make_empty(name, loc, parent=None, size=0.4): e = bpy.data.objects.new(name, None) e.empty_display_type = 'PLAIN_AXES' e.empty_display_size = size e.location = loc coll.objects.link(e) if parent is not None: e.parent = parent return e def cylinder(name, radius, depth, loc=(0, 0, 0), rot=(0, 0, 0), mat=None, parent=None, verts=32): bpy.ops.mesh.primitive_cylinder_add(vertices=verts, radius=radius, depth=depth, location=loc, rotation=rot) o = bpy.context.active_object o.name = name return register(o, parent, mat) def cone(name, r_base, r_tip, depth, loc, rot, mat=None, parent=None): bpy.ops.mesh.primitive_cone_add(vertices=32, radius1=r_base, radius2=r_tip, depth=depth, location=loc, rotation=rot) o = bpy.context.active_object o.name = name return register(o, parent, mat) def box(name, size, loc, mat=None, parent=None): bpy.ops.mesh.primitive_cube_add(size=1, location=loc) o = bpy.context.active_object o.name = name o.scale = size return register(o, parent, mat) def ellipsoid(name, scale, loc, mat=None, parent=None): bpy.ops.mesh.primitive_uv_sphere_add(segments=32, ring_count=16, radius=1, location=loc) o = bpy.context.active_object o.name = name o.scale = scale return register(o, parent, mat, smooth=True) def loft(name, rings, mat=None, parent=None, smooth=False): """Build a closed solid by bridging a list of vertex rings (each ring = list of Vectors).""" mesh = bpy.data.meshes.new(name + "_mesh") bm = bmesh.new() vrings = [[bm.verts.new(p) for p in ring] for ring in rings] n = len(rings[0]) for a, b in zip(vrings, vrings[1:]): for i in range(n): j = (i + 1) % n bm.faces.new((a[i], a[j], b[j], b[i])) bm.faces.new(vrings[0]) # end caps bm.faces.new(vrings[-1]) bmesh.ops.recalc_face_normals(bm, faces=bm.faces[:]) bm.to_mesh(mesh) bm.free() obj = bpy.data.objects.new(name, mesh) return register(obj, parent, mat, smooth) # ------------------------------------------------------------ root + fuselage root = make_empty("Aircraft_Root", (0, 0, 0), size=1.0) # (x, radius, z-centre) stations, listed from nose (+X) to tail (-X) FUSELAGE = [ (10.00, 0.04, -0.15), (9.75, 0.40, -0.12), (9.20, 0.78, -0.06), (8.40, 1.08, -0.02), (7.00, 1.27, 0.00), (5.50, 1.32, 0.00), (-4.50, 1.32, 0.00), (-6.50, 1.22, 0.05), (-8.00, 0.95, 0.22), (-9.40, 0.55, 0.36), (-10.20, 0.12, 0.42), ] SEG = 32 fus_rings = [] for x, r, zc in FUSELAGE: fus_rings.append([ Vector((x, r * math.cos(2 * math.pi * k / SEG), zc + r * math.sin(2 * math.pi * k / SEG))) for k in range(SEG) ]) loft("Fuselage", fus_rings, M_BODY, root, smooth=True) # cockpit windshield ellipsoid("Windshield", (0.85, 0.75, 0.40), (8.75, 0, 0.62), M_GLASS, root) # passenger windows (two rows of small discs) x = -4.0 while x <= 5.0: for side in (1, -1): cylinder("Window", 0.11, 0.08, loc=(x, 1.27 * side, 0.35), rot=(math.pi / 2, 0, 0), mat=M_GLASS, parent=root, verts=16) x += 0.7 # ------------------------------------------------------------------- wings # airfoil outline as (fraction of chord back from leading edge, fraction of thickness) AIRFOIL = [(0.0, 0.0), (0.2, 0.5), (0.6, 0.42), (1.0, 0.0), (0.6, -0.30), (0.2, -0.45)] def wing_ring(y, x_le, chord, zc, thick): return [Vector((x_le - fx * chord, y, zc + fz * thick)) for fx, fz in AIRFOIL] for side, nm in ((1, "L"), (-1, "R")): rings = [ wing_ring(1.0 * side, 1.5, 4.6, -0.30, 0.50), # root wing_ring(10.5 * side, -3.0, 1.5, 0.10, 0.12), # tip (swept back, slight dihedral) ] loft("Wing_" + nm, rings, M_WING, root, smooth=False) # horizontal stabilizers for side, nm in ((1, "L"), (-1, "R")): rings = [ wing_ring(0.5 * side, -7.6, 2.4, 0.30, 0.20), wing_ring(4.2 * side, -9.4, 1.1, 0.50, 0.08), ] loft("HStab_" + nm, rings, M_WING, root) # vertical fin (sections lie in the XY plane at a given height) FIN = [(0.0, 0.0), (0.25, 0.5), (0.65, 0.4), (1.0, 0.0), (0.65, -0.4), (0.25, -0.5)] def fin_ring(z, x_le, chord, thick): return [Vector((x_le - fx * chord, fy * thick, z)) for fx, fy in FIN] loft("VFin", [fin_ring(0.6, -5.6, 3.4, 0.28), fin_ring(4.8, -8.8, 1.6, 0.10)], M_TAIL, root) # ----------------------------------------------------------------- engines for side, nm in ((1, "L"), (-1, "R")): y = 4.2 * side cx, cz, length, rad = 0.5, -1.05, 3.2, 0.75 cylinder("Engine_" + nm, rad, length, loc=(cx, y, cz), rot=(0, math.pi / 2, 0), mat=M_ENGINE, parent=root) cone("Exhaust_" + nm, rad, 0.35, 0.9, loc=(cx - length / 2 - 0.45, y, cz), rot=(0, -math.pi / 2, 0), mat=M_ENGINE, parent=root) cylinder("Intake_" + nm, rad * 0.82, 0.06, loc=(cx + length / 2 + 0.005, y, cz), rot=(0, math.pi / 2, 0), mat=M_GLASS, parent=root) box("Pylon_" + nm, (1.8, 0.18, 0.55), (cx + 0.1, y, -0.45), M_WING, root) # -------------------------------------------------------- landing gear rig def build_gear(name, hinge, strut_len, wheel_r, wheel_w, wheel_offsets, strut_r): """Strut + wheels hung from a hinge Empty. Rotate the Empty to retract.""" pivot = make_empty(name + "_Hinge", hinge, root, size=0.3) cylinder(name + "_Strut", strut_r, strut_len, loc=(0, 0, -strut_len / 2), mat=M_METAL, parent=pivot, verts=16) span = (max(wheel_offsets) - min(wheel_offsets)) + 0.1 cylinder(name + "_Axle", 0.05, span, loc=(0, 0, -strut_len), rot=(math.pi / 2, 0, 0), mat=M_METAL, parent=pivot, verts=12) for i, off in enumerate(wheel_offsets): cylinder("%s_Tire%d" % (name, i), wheel_r, wheel_w, loc=(0, off, -strut_len), rot=(math.pi / 2, 0, 0), mat=M_TIRE, parent=pivot) cylinder("%s_Rim%d" % (name, i), wheel_r * 0.55, wheel_w + 0.02, loc=(0, off, -strut_len), rot=(math.pi / 2, 0, 0), mat=M_METAL, parent=pivot) return pivot # Ground is GROUND_Z: hinge_z - strut_len - wheel_r == -2.6 for every gear leg. nose = build_gear("NoseGear", (6.2, 0, -0.6), 1.7, 0.30, 0.20, (-0.17, 0.17), 0.08) main_l = build_gear("MainGear_L", (-1.5, 2.3, -0.3), 1.8, 0.50, 0.32, (-0.22, 0.22), 0.11) main_r = build_gear("MainGear_R", (-1.5, -2.3, -0.3), 1.8, 0.50, 0.32, (-0.22, 0.22), 0.11) # ---------------------------------------------------------- gear animation UP = math.radians(90) def key_rot(obj, frame, rot): obj.rotation_euler = rot obj.keyframe_insert(data_path="rotation_euler", frame=frame) # nose gear folds FORWARD (rotate about Y) for f, a in ((1, 0), (20, 0), (55, -UP), (110, -UP), (145, 0), (160, 0)): key_rot(nose, f, (0, a, 0)) # main gear fold INWARD toward the fuselage (rotate about X, opposite signs per side) for f, a in ((1, 0), (25, 0), (65, UP), (105, UP), (145, 0), (160, 0)): key_rot(main_l, f, (-a, 0, 0)) key_rot(main_r, f, (a, 0, 0)) scene = bpy.context.scene scene.frame_start = FRAME_START scene.frame_end = FRAME_END scene.frame_set(FRAME_START) # ------------------------------------------------------ ground, light, camera if ADD_GROUND: bpy.ops.mesh.primitive_plane_add(size=120, location=(0, 0, GROUND_Z)) g = bpy.context.active_object g.name = "Ground" register(g, None, M_GROUND) sun_data = bpy.data.lights.new("Sun", 'SUN') sun_data.energy = 3.5 sun = bpy.data.objects.new("Sun", sun_data) sun.rotation_euler = (math.radians(50), math.radians(10), math.radians(35)) coll.objects.link(sun) target = make_empty("Camera_Target", (0, 0, -0.5), size=0.3) cam_data = bpy.data.cameras.new("Camera") cam = bpy.data.objects.new("Camera", cam_data) cam.location = (24, -24, 8) coll.objects.link(cam) track = cam.constraints.new('TRACK_TO') track.target = target track.track_axis = 'TRACK_NEGATIVE_Z' track.up_axis = 'UP_Y' scene.camera = cam if SAVE_BLEND_TO: bpy.ops.wm.save_as_mainfile(filepath=SAVE_BLEND_TO) print("Aircraft built. Press SPACE in the viewport to play the gear retract/extend cycle.")
